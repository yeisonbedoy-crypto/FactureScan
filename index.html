Conectar con Google Sheets — 3 pasos
1
Prepara tu hoja de Google Sheets
Ve a sheets.new y crea una hoja nueva. En la primera fila añade estas cabeceras (una por columna, de A a Q):
A · fecha_escaneo B · fecha C · numero_factura D · proveedor E · nif_proveedor F · concepto G · base_imponible H · tipo_impuesto I · porcentaje_impuesto J · cuota_impuesto K · total L · moneda M · tipo_contable N · forma_pago O · cliente P · nif_cliente Q · observaciones
Pega en celda A1 → Datos → Dividir texto en columnas (separador: tabulación).
2
Pega el script en Apps Script
En tu hoja → Extensiones → Apps Script → borra el código existente y pega este:
function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getActiveSheet(); var cb = e.parameter.callback || ''; var action = e.parameter.action || 'write'; // ── READ ACTION: return dashboard data ── if (action === 'read') { var lastRow = sheet.getLastRow(); if (lastRow < 2) { var empty = JSON.stringify({status:'ok',mes:{total:0,impuestos:0,base:0,count:0},anual:{total:0,impuestos:0,base:0,count:0,devoluciones:0},recientes:[],meses:[]}); return ContentService.createTextOutput(cb ? cb+'('+empty+')' : empty) .setMimeType(cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON); } var now = new Date(); var mesActual = now.getMonth(); var anioActual = now.getFullYear(); // Parse filter dates var dateFromStr = e.parameter.dateFrom || null; var dateToStr = e.parameter.dateTo || null; var dateFromObj = dateFromStr ? new Date(dateFromStr + 'T00:00:00Z') : null; var dateToObj = dateToStr ? new Date(dateToStr + 'T23:59:59Z') : null; // Read all data columns: B=fecha(1), C=numero_factura(2), D=proveedor(3), // G=base(6), J=cuota(9), K=total(10), M=tipo_contable(12) var data = sheet.getRange(2, 1, lastRow - 1, 17).getValues(); var mes = { total:0, impuestos:0, base:0, count:0 }; var anual = { total:0, impuestos:0, base:0, count:0, devoluciones:0 }; var recientes = []; var mesesMap = {}; for (var i = 0; i < data.length; i++) { var row = data[i]; var fecha = row[1]; // col B var numFac = row[2]; // col C var prov = row[3]; // col D var base = parseFloat(row[6]) || 0; // col G var cuota = parseFloat(row[9]) || 0; // col J var total = parseFloat(row[10]) || 0; // col K var tipo = row[12] || ''; // col M if (!fecha && !total) continue; // skip empty rows // Parse date var d = fecha instanceof Date ? fecha : new Date(fecha); var rowMes = d.getMonth ? d.getMonth() : -1; var rowAnio = d.getMonth ? d.getFullYear(): -1; // Check if row is within date filter (if specified) var withinDateRange = true; if (dateFromObj && d < dateFromObj) withinDateRange = false; if (dateToObj && d > dateToObj) withinDateRange = false; // Annual totals (always calculate for KPIs) if (rowAnio === anioActual) { anual.count++; if (total < 0) { anual.devoluciones += Math.abs(total); } else { anual.total += total; anual.impuestos += cuota; anual.base += base; } // Monthly bar chart (last 6 months) var mesKey = rowAnio + '-' + String(rowMes + 1).padStart(2,'0'); if (!mesesMap[mesKey]) mesesMap[mesKey] = 0; mesesMap[mesKey] += total; } // Current month totals if (rowMes === mesActual && rowAnio === anioActual && total >= 0) { mes.count++; mes.total += total; mes.impuestos += cuota; mes.base += base; } // Recent invoices - include ALL if filter is active, otherwise last 8 if (withinDateRange) { recientes.push({ fecha: fecha instanceof Date ? Utilities.formatDate(fecha, Session.getScriptTimeZone(), 'dd/MM/yyyy') : String(fecha).substring(0,10), proveedor: String(prov).substring(0,25), numero: String(numFac), tipo: String(tipo), total: total }); } } // If no date filter, limit to last 8; if filter active, return all matching if (!dateFromStr && !dateToStr) { recientes = recientes.slice(-8).reverse(); } else { recientes = recientes.reverse(); // Just reverse, don't limit } // Build last 6 months array var mesesArr = []; for (var m = 5; m >= 0; m--) { var d2 = new Date(anioActual, mesActual - m, 1); var key = d2.getFullYear() + '-' + String(d2.getMonth()+1).padStart(2,'0'); var nombres = ['Ene','Feb','Mar','Abr','May','Jun','Jul','Ago','Sep','Oct','Nov','Dic']; mesesArr.push({ mes: nombres[d2.getMonth()], total: mesesMap[key] || 0, isCurrent: m === 0 }); } // Round all values function r2(n) { return Math.round(n * 100) / 100; } mes.total = r2(mes.total); mes.impuestos = r2(mes.impuestos); mes.base = r2(mes.base); anual.total = r2(anual.total); anual.impuestos = r2(anual.impuestos); anual.base = r2(anual.base); anual.devoluciones = r2(anual.devoluciones); anual.total = r2(anual.total - anual.devoluciones); // net of returns var result = JSON.stringify({status:'ok', mes:mes, anual:anual, recientes:recientes, meses:mesesArr}); return ContentService.createTextOutput(cb ? cb+'('+result+')' : result) .setMimeType(cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON); } // ── WRITE ACTION (default) ── // Auto-headers si la hoja está vacía — nuevo orden if (sheet.getLastRow() === 0) { sheet.appendRow([ "fecha_escaneo","fecha","numero_factura", "proveedor","nif_proveedor","concepto", "base_imponible","tipo_impuesto","porcentaje_impuesto", "cuota_impuesto","total","moneda", "tipo_contable","forma_pago","cliente", "nif_cliente","observaciones" ]); } // Normaliza: elimina todo lo que no sea alfanumérico function norm(s) { return String(s || '').replace(/[^a-zA-Z0-9]/g, '').toLowerCase(); } // DUPLICATE CHECK: numero_factura en Columna C (índice 3) var numFacturaNorm = norm(e.parameter.numero_factura); if (numFacturaNorm && sheet.getLastRow() > 1) { var colC = sheet.getRange(2, 3, sheet.getLastRow() - 1, 1).getValues(); for (var i = 0; i < colC.length; i++) { if (norm(colC[i][0]) === numFacturaNorm) { var dupJson = JSON.stringify({status:"error", msg:"DUPLICATE"}); var dupOut = cb ? cb + '(' + dupJson + ')' : dupJson; return ContentService.createTextOutput(dupOut) .setMimeType(cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON); } } } function toNum(s) { if (!s || s === '') return ''; var clean = String(s).trim(); if (clean.indexOf(',') > -1 && clean.indexOf('.') > -1) { if (clean.lastIndexOf(',') > clean.lastIndexOf('.')) { clean = clean.replace(/\./g,'').replace(',','.'); } else { clean = clean.replace(/,/g,''); } } else { clean = clean.replace(',','.'); } var n = parseFloat(clean); return isNaN(n) ? s : n; } var p = e.parameter; sheet.appendRow([ p.fecha_escaneo || new Date().toLocaleString("es-ES"), // A p.fecha || "", // B p.numero_factura || "", // C p.proveedor || "", // D p.nif_proveedor || "", // E p.concepto || "", // F toNum(p.base_imponible), // G p.tipo_impuesto || "", // H toNum(p.porcentaje_impuesto), // I toNum(p.cuota_impuesto), // J toNum(p.total), // K p.moneda || "EUR", // L p.tipo_contable || "", // M p.forma_pago || "", // N p.cliente || "", // O p.nif_cliente || "", // P p.observaciones || "" // Q ]); var json = JSON.stringify({status:"ok"}); var out = cb ? cb + '(' + json + ')' : json; return ContentService.createTextOutput(out) .setMimeType(cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON); } catch(err) { var json = JSON.stringify({status:"error", msg:err.message}); var out = (e.parameter.callback || '') + '(' + json + ')'; return ContentService.createTextOutput(out) .setMimeType(ContentService.MimeType.JAVASCRIPT); } }

📐 Fórmulas para totales en Sheets
Total IVA/IGIC
=SUMIF(M:M,"IVA/IGIC",K:K)
Total exentas
=SUMIF(M:M,"EXENTA",K:K)
Total general
=SUM(K2:K)
Total cuota impuesto
=SUM(J2:J)
Recuento facturas
=COUNTA(C2:C)
💡 Consejo: Crea una pestaña "Resumen" en tu Sheets y pon ahí estas fórmulas apuntando a la hoja de datos.
